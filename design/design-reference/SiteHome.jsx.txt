function HeroLede({ t }) {
  const fmt = (t && t.ledeFormat) || 'Paragraph';
  const size = (t && t.ledeSize) || 19;
  const width = (t && t.ledeWidth) || 640;
  const st = { fontSize: size, maxWidth: width };
  const intro = 'System Switch is a growth infrastructure firm for founder-led coaches, consultants and service providers.';
  if (fmt === 'Split') return (
    <div className="s-lede-split" style={{ maxWidth: width }}>
      <p className="s-lede-a" style={{ fontSize: size + 2 }}>{intro}</p>
      <p className="s-lede-b" style={{ fontSize: size - 2 }}>We find where revenue is leaking, install the systems that stop it, and hand you the controls.</p>
    </div>
  );
  if (fmt === 'Three steps') return (
    <div className="s-lede-steps" style={{ maxWidth: Math.max(width, 820) }}>
      <p className="ss-body-lg s-lede" style={st}>{intro}</p>
      <ol>{['We find where revenue is leaking,', 'install the systems that stop it,', 'and hand you the controls.'].map((x, i) => (
        <li key={x}><span className="s-qn">{String(i + 1).padStart(2, '0')}</span><span>{x}</span></li>
      ))}</ol>
    </div>
  );
  if (fmt === 'Ruled') return (
    <div className="s-lede-ruled" style={{ maxWidth: width }}>
      <p style={{ fontSize: size }}>{intro} We find where revenue is leaking, install the systems that stop it, and hand you the controls.</p>
    </div>
  );
  return <p className="ss-body-lg s-lede" style={st}>{intro} We find where revenue is leaking, install the systems that stop it, and hand you the controls.</p>;
}
