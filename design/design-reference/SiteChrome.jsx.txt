function CookieBanner() {
  const [shown, setShown] = React.useState(localStorage.getItem('ssd-site-cookies') !== 'set' && !location.search.includes('review'));
  if (!shown) return null;
  const done = () => { localStorage.setItem('ssd-site-cookies', 'set'); setShown(false); };
  return (
    <div className="s-cookie" role="region" aria-label="Cookies">
      <div className="s-wrap s-cookie-in">
        <p className="ss-body-sm">We use essential cookies to run this site, and optional ones to understand how it's used.</p>
        <div className="s-cookie-btns"><SButton variant="secondary" size="sm" onClick={done}>Accept all</SButton><SButton variant="secondary" size="sm" onClick={done}>Essential only</SButton></div>
      </div>
    </div>
  );
}
