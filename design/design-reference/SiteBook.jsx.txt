function QLabel({ n, q }) {
  return <React.Fragment><span className="s-qn">{String(n).padStart(2, '0')}</span>{q.label}{q.optional ? <span className="s-opt"> (optional)</span> : null}</React.Fragment>;
}

function Radios({ q, n, value, onChange, invalid }) {
  return (
    <fieldset className={'s-radios' + (invalid ? ' s-invalid' : '')}>
      <legend className="s-qlabel"><QLabel n={n} q={q} /></legend>
      <div className="s-radio-list">{q.opts.map((o) => (
        <label key={o} className="s-radio"><input type="radio" name={q.id} value={o} checked={value === o} onChange={() => onChange(o)} /><span>{o}</span></label>
      ))}</div>
    </fieldset>
  );
}

function LongField({ q, n, value, onChange, invalid }) {
  return (
    <div className={'ss-field s-long' + (invalid ? ' s-invalid' : '')}>
      <label className="s-qlabel" htmlFor={'q-' + q.id}><QLabel n={n} q={q} /></label>
      <textarea id={'q-' + q.id} className="ss-field-input" rows={4} value={value || ''} onChange={(e) => onChange(e.target.value)} aria-invalid={invalid ? 'true' : undefined}></textarea>
      {q.help ? <span className="ss-field-help">{q.help}</span> : null}
    </div>
  );
}

function NextPanel({ step }) {
  const rows = ['Tell us about your business', 'Choose a time', "We'll show you where we'd start"];
  return (
    <aside className="s-next">
      <span className="ss-label">What happens next</span>
      <ol>{rows.map((r, i) => <li key={r} aria-current={i === step ? 'step' : undefined}><span className="s-qn">{String(i + 1).padStart(2, '0')}</span>{r}</li>)}</ol>
    </aside>
  );
}

function BookForm({ go }) {
  const [v, setV] = React.useState({});
  const [errs, setErrs] = React.useState(null);
  const set = (id) => (val) => setV((s) => ({ ...s, [id]: val }));
  const submit = (e) => {
    e.preventDefault();
    const bad = {};
    BOOK_SECTIONS.forEach((s) => s.qs.forEach((q) => { if (!q.optional && !String(v[q.id] || '').trim()) bad[q.id] = true; }));
    if (Object.keys(bad).length) { setErrs(bad); window.scrollTo({ top: 0 }); return; }
    go('start/calendar');
  };
  let n = 0;
  return (
    <form className="s-form" onSubmit={submit} noValidate>
      {errs ? <p className="s-form-error ss-body" role="alert">Something didn't go through. Check the highlighted fields and try again.</p> : null}
      {BOOK_SECTIONS.map((s) => (
        <section key={s.title} className="ss-card s-form-card">
          <h2 className="ss-h3" style={{ margin: 0 }}>{s.title}</h2>
          {s.qs.map((q) => {
            n += 1;
            const inv = errs && errs[q.id];
            if (q.opts) return <Radios key={q.id} q={q} n={n} value={v[q.id]} onChange={set(q.id)} invalid={inv} />;
            if (q.long) return <LongField key={q.id} q={q} n={n} value={v[q.id]} onChange={set(q.id)} invalid={inv} />;
            return <BField key={q.id} id={'q-' + q.id} className={inv ? 's-invalid' : ''} label={<QLabel n={n} q={q} />} type={q.type} autoComplete={q.auto} help={q.help} value={v[q.id] || ''} onChange={(e) => set(q.id)(e.target.value)} aria-invalid={inv ? 'true' : undefined} />;
          })}
        </section>
      ))}
      <div><SButton variant="primary" type="submit">Choose a time →</SButton></div>
    </form>
  );
}
